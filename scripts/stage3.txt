# Подходит для любой VFS (и для запуска без --vfs).
# Запуск: run.bat --vfs vfs\deep.csv --script scripts\stage3.txt

vfsinfo             # служебная команда: имя VFS, число папок и файлов
ls                  # содержимое корня загруженной VFS
ls -l               # то же, подробно
cd /                # переход в корень
cd                  # без аргументов - тоже в корень
vfsinfo extra       # ошибка: у служебной команды нет аргументов
ls                  # эта строка уже не выполнится
