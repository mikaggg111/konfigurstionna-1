# Стартовый скрипт этапа 4: ls, cd, tac, clear, uptime.
# Рассчитан на vfs\deep.csv и останавливается на первой ошибке.
# Запуск: run.bat --vfs vfs\deep.csv --script scripts\stage4.txt
# Остальные ошибки проверяются по одной: scripts\errors\*.txt

vfsinfo                         # какая VFS загружена

# --- ls: все режимы ---
ls                              # корень, у папок в конце слэш
ls -l                           # подробно: тип, размер, имя
ls /home/user                   # путь к папке
ls /home/user/notes.txt         # путь к файлу
ls /etc /empty                  # несколько папок сразу
ls -l /home/user/docs/archive/2025

# --- cd: все режимы ---
cd /home/user/docs              # абсолютный путь (в приглашении папка)
ls
cd archive/2025                 # относительный путь
ls
cd ../..                        # два шага вверх
cd .                            # текущая папка
cd /                            # в корень
cd home                         # относительно текущей папки
cd user                         # ещё глубже
cd ..                           # на шаг вверх
cd                              # без аргументов - в корень

# --- tac: все режимы ---
tac /home/user/notes.txt        # строки в обратном порядке
cd /home/user
tac notes.txt docs/todo.txt     # несколько файлов, относительные пути
tac /etc/motd

# --- uptime: все режимы ---
uptime                          # время, сколько работает эмулятор
uptime -p                       # то же словами
uptime -s                       # когда эмулятор запущен

# --- clear: очищает экран, скрипт продолжается ---
clear
ls                              # после clear на экране только эта команда
tac /no/such/file.txt           # ошибка: скрипт остановится
ls                              # эта строка уже не выполнится
