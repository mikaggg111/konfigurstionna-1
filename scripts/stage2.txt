# Стартовый скрипт этапа 2.
# Строка, начинающаяся с #, - комментарий. Пустые строки пропускаются.
# На экране видны и команды, и их вывод. Скрипт останавливается на
# первой ошибке. Подходит для запуска без VFS и с любой VFS.

ls                                       # команда без аргументов
ls -l                                    # команда с опцией
cd /                                     # переход в корень
cd .                                     # "." - текущая папка

exit now                                 # ошибка: exit не принимает аргументы
ls                                       # эта строка уже не выполнится
